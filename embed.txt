# paraphrase-multilingual-mpnet-base-v2: 한국어 지원 및 768차원 고정 출력 모델

def fill_missing_embeddings():
    """임베딩이 비어 있는(null) 데이터를 찾아 벡터를 생성하고 업데이트합니다."""
    print("\n[*] 데이터베이스에서 임베딩이 누락된 항목을 조회 중...")
    try:
        # embedding이 null인 행 가져오기
        res = supabase_client.table("pdf_knowledge").select("id, content").is_("embedding", "null").execute()
        missing_data = res.data

        if not missing_data:
            print("[✔] 모든 데이터에 임베딩이 채워져 있습니다.")
            return

        print(f"[!] 총 {len(missing_data)}개의 누락된 임베딩을 발견했습니다. 작업을 시작합니다...")

        batch_size = 50
        for i in range(0, len(missing_data), batch_size):
            batch = missing_data[i : i + batch_size]
            
            for item in batch:
                try:
                    # 로컬 모델로 벡터 생성
                    vector = embed_model.encode(item['content']).tolist()
                    
                    # 해당 행 업데이트
                    supabase_client.table("pdf_knowledge").update({
                        "embedding": vector
                    }).eq("id", item['id']).execute()
                except Exception as e:
                    print(f"    - [!] ID {item['id']} 처리 실패: {e}")

            progress = min(i + batch_size, len(missing_data))
            print(f"    - 최적화 진행률: {progress}/{len(missing_data)} 완료")

        print("[✅ 완료] 누락된 모든 임베딩이 복구되었습니다.")
    except Exception as e:
        print(f"[❌ 오류] 임베딩 복구 중 문제 발생: {e}")

def process_pdf_with_ocr(file_path, project_id="GENERAL"):
    raw_name = os.path.basename(file_path)
    # 파일명 정규화: 브라우저나 OS가 붙이는 " (1)", "(1)" 등의 접미사를 제거하여 파일명을 통일합니다.
    # 예: "지침 (1).pdf" -> "지침.pdf"
    file_name = re.sub(r"\s?\(\d+\)(?=\.[^.]+$)", "", raw_name)

    print(f"\n[*] '{raw_name}' 분석 시작 (저장명: '{file_name}')...")

    # 1. 기존 데이터 삭제 (파일명 기준 중복 방지)
    print(f"[*] 기존 데이터 정리 중...")
    supabase_client.table("pdf_knowledge").delete().eq("file_name", file_name).execute()

    # 2. PDF 텍스트 추출
    reader = PdfReader(file_path)
    all_chunks = []
    chunk_size = 700

    for i, page in enumerate(reader.pages):
        text = page.extract_text()
        
        # [OCR 로직 구현] 텍스트가 없거나 너무 적으면 이미지로 변환하여 OCR 실행
        if not text or len(text.strip()) < 10:
            print(f"    - [{i+1}페이지] 이미지 감지: OCR 분석 중...")
            try:
                # 해당 페이지를 이미지로 변환 (300 DPI)
                images = convert_from_path(file_path, first_page=i+1, last_page=i+1)
                if images:
                    img_np = np.array(images[0])
                    ocr_results = reader_ocr.readtext(img_np, detail=0)
                    text = " ".join(ocr_results)
            except Exception as ocr_err:
                print(f"    - [!] {i+1}페이지 OCR 실패: {ocr_err}")
        
        if not text: continue
        
        clean_text = " ".join(text.split())
        for j in range(0, len(clean_text), chunk_size):
            chunk_content = clean_text[j : j + chunk_size]
            all_chunks.append({"page": i + 1, "text": chunk_content})

    if not all_chunks:
        print("[!] 추출된 내용이 없습니다. 파일을 확인해주세요.")
        return

    print(f"[+] 총 {len(all_chunks)}개의 데이터 조각 생성 완료. 벡터 계산 시작...")

    # 3. 임베딩 생성 및 일괄 저장
    batch_size = 50 # 한 번에 50개씩 저장
    batch_data = []
    
    for idx, chunk in enumerate(all_chunks, 1): # idx를 1부터 시작
        try:
            embedding = embed_model.encode(chunk["text"]).tolist()
            batch_data.append({
                "project_id": project_id,
                "file_name": file_name,
                "content": chunk["text"],
                "embedding": embedding,
                "metadata": {"page": chunk["page"]}
            })
            
            if idx % batch_size == 0 or idx == len(all_chunks):
                print(f"    - [{idx}/{len(all_chunks)}] 배치 저장 중...")
                supabase_client.table("pdf_knowledge").insert(batch_data).execute()
                batch_data = [] # 배치 초기화
        
        except Exception as e:
            print(f"    - [!] {idx}번째 조각 처리 중 오류: {e}")
            # 오류 발생 시 해당 조각만 건너뛰고 다음 조각 처리 (선택 사항)
            # 또는 오류 발생 시 남은 배치 데이터도 저장 시도 후 종료
            if batch_data:
                try:
                    supabase_client.table("pdf_knowledge").insert(batch_data).execute()
                    print(f"    - [!] 오류 발생 전 {len(batch_data)}개 조각 저장 완료.")
                except Exception as batch_e:
                    print(f"    - [!] 남은 배치 저장 중 추가 오류: {batch_e}")
            batch_data = [] # 배치 초기화

    print(f"\n[✅ 성공] '{file_name}' 데이터가 Supabase에 저장되었습니다.")
    print("[*] 로컬 엔진(Sentence-Transformers)을 사용했으므로 API 할당량 소모가 없으며 768차원으로 고정되었습니다.")
