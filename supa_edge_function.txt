import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent`;
// 사용자님의 모델 리스트 35번 항목(gemini-embedding-001)으로 모델 변경
const EMBED_URL = `${API_BASE}/models/gemini-embedding-001:embedContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // [추가] AI가 사용할 수 있는 도구(함수) 정의
    const tools = [
      {
        function_declarations: [
          {
            name: "query_table_data",
            description: "프로젝트 정보(상세, 메모, 사진, 도면파일)를 조회합니다. 명칭이나 장소로 질문 시 반드시 'cad_projects' 테이블에서 'name' 컬럼으로 ID를 먼저 검색하세요.",
            parameters: {
              type: "OBJECT",
              properties: {
                table: { type: "STRING", enum: ["cad_projects", "project_detail", "memos", "photos", "cad_files"], description: "조회할 테이블명" },
                project_id: { type: "STRING", description: "숫자로 된 프로젝트 고유 ID" },
                search_keyword: { type: "STRING", description: "프로젝트 한글 이름 검색 키워드 (cad_projects 테이블 조회 시 사용)" }
              },
              required: ["table"]
            }
          },
          {
            name: "execute_web_command",
            description: "웹 화면을 조작하거나 특정 기능을 실행합니다 (지도 이동, 탭 전환 등).",
            parameters: {
              type: "OBJECT",
              properties: {
                action: { type: "STRING", enum: ["open_map", "switch_tab", "search_point"], description: "실행할 액션" },
                target_id: { type: "STRING", description: "대상 프로젝트 ID 또는 탭 이름" }
              },
              required: ["action"]
            }
          }
        ]
      }
    ];

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = `당신은 '아신테크 GIS 관리 도구'의 전문 에이전트입니다.
[데이터 접근 규칙]
1. 사용자 질문에 포함된 지명이나 현장 이름은 'cad_projects' 테이블의 'name' 컬럼에 해당합니다. 
2. 반드시 'cad_projects'에서 ID를 먼저 찾은 후, 그 ID를 사용하여 'project_detail', 'memos' 등에서 상세 정보를 조회하세요.
3. 'project_detail'의 'work_location'은 검색 기준으로 사용하지 마세요.
4. 보안 경고: 'project_shares', 'user_settings' 테이블은 절대 접근 금지입니다.
5. 필요한 정보(ID 등)를 얻었다면 멈추지 말고 다음 단계(메모 조회 등)의 도구를 계속 호출하여 최종 결론을 내세요.
6. 데이터가 없는 경우 절대 지어내지 말고 "데이터를 찾을 수 없다"고 답변하세요.`;
    let vectorContext = "";

    // [복구] 지침서 기반 답변을 위한 벡터 검색 로직
    if (type === 'point_search' && !prompt.includes("프로젝트")) { // 단순 검색일 때만 벡터 검색 수행
        // 1. 질문의 임베딩 생성 (검색용 - gemini-embedding-001 사용)
        const embedRes = await fetch(`${EMBED_URL}?key=${GEMINI_API_KEY}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ 
                content: { parts: [{ text: prompt }] }, 
                task_type: "RETRIEVAL_QUERY"
            })
        });
        const embedData = await embedRes.json();
        const queryEmbedding = embedData.embedding?.values;

        if (queryEmbedding) {
            // 2. Supabase RPC 호출하여 유사 지식 검색
            const searchRes = await fetch(`${SUPABASE_URL}/rest/v1/rpc/match_pdf_knowledge`, {
                method: 'POST',
                headers: {
                    'apikey': SUPABASE_KEY,
                    'Authorization': `Bearer ${SUPABASE_KEY}`,
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    query_embedding: queryEmbedding,
                    match_threshold: 0.3, // 유사도 임계값 조절 가능
                    match_count: 5
                })
            });

            if (searchRes.ok) {
                const matchedKnowledge = await searchRes.json();
                if (matchedKnowledge && matchedKnowledge.length > 0) {
                    vectorContext = "\n\n[참고 지침서 내용]\n" + matchedKnowledge.map(k => k.content).join("\n---\n");
                    systemInstruction += "제공된 '[참고 지침서 내용]'을 최우선 근거로 사용하여 답변하세요. 수치나 규격 등 비교가 필요한 데이터는 반드시 마크다운 표(|---|)를 사용하여 가독성을 높이세요.";
                }
            }
        }
        systemInstruction += "\n사용자가 입력한 검색어가 GIS 관련 용어라면 제공된 도면 레이어 정보를 참고하여 답변하고, 만약 일반적인 질문이라면 범용적인 AI로서 답변하세요.";
    } else if (type === 'pdf_summary') {
      systemInstruction += "제공된 PDF 텍스트 내용을 요약하고 사용자의 질문에 답하세요.";
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000);

    // [핵심] 에이전틱 루프 구현: AI가 도구를 다 쓸 때까지 최대 5회 반복
    let contents = [{
      role: "user",
      parts: [{ text: `${systemInstruction}\n\n[도면 레이어 맥락]\n${context || ""}${vectorContext}\n\n[User Query]\n${prompt}` }]
    }];
    let finalAnswer = "";
    let lastRawData = [];
    let loopCount = 0;
    let data = null; // data 변수를 loop 외부에서 접근 가능하도록 상단에 선언

    while (loopCount < 5) {
      const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        signal: controller.signal,
        body: JSON.stringify({
          contents: contents,
          tools: tools,
          generationConfig: { temperature: 0.2, maxOutputTokens: 2048 }
        }),
      });

      data = await response.json(); // const를 제거하여 외부 변수 data를 업데이트
      if (data.error) throw new Error(data.error.message);

      const candidate = data.candidates?.[0];
      const part = candidate?.content?.parts?.[0];

      // 1. 함수 호출 요청이 있는 경우
      if (part?.functionCall) {
        const call = part.functionCall;
        let result = [];

        if (call.name === "query_table_data") {
          const { table, project_id, search_keyword } = call.args;
          let url = `${SUPABASE_URL}/rest/v1/${table}?select=*`;
          if (project_id) {
            const idField = (table === 'photos') ? 'cad_project_id' : (table === 'cad_projects' ? 'id' : 'project_id');
            url += `&${idField}=eq.${project_id}`;
          }
          if (search_keyword) {
            let nameField = (table === 'cad_projects') ? 'name' : (table === 'memos' ? 'content' : 'file_name');
            url += `&${nameField}=ilike.*${encodeURIComponent(search_keyword)}*`;
          }
          const dbRes = await fetch(url, { headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}` } });
          result = await dbRes.json();
          lastRawData = result; // 마지막 조회 데이터를 저장
        }
        
        if (call.name === "execute_web_command") {
          return new Response(JSON.stringify({ success: true, answer: "명령 실행", command: call.args }), { headers: corsHeaders });
        }

        // 대화 기록에 추가하고 루프 계속
        contents.push(candidate.content); // 모델의 function_call 추가
        contents.push({
          role: "function",
          parts: [{ functionResponse: { name: call.name, response: { content: result } } }]
        });
        loopCount++;
      } 
      // 2. 최종 텍스트 답변이 온 경우
      else if (part?.text) {
        finalAnswer = part.text;
        break;
      } 
      else {
        break;
      }
    }

    clearTimeout(timeoutId);

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer || "답변을 생성할 수 없습니다.",
        rawData: JSON.stringify(lastRawData),
        model: data.model_version || "gemini-2.5-flash-lite"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (30초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: isTimeout ? 504 : 400 
      }
    );
  }
})
