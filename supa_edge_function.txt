import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
const API_URL = `${API_BASE}/models/gemini-2.5-flash:generateContent`;
const EMBED_URL = `${API_BASE}/models/text-embedding-004:embedContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = "당신은 '아신테크 GIS 관리 도구'의 전문 어시스턴트입니다. 사용자의 질문에 친절하고 정확하게 답변하세요. \n\n[답변 원칙]\n1. 지침서 원문에 포함된 페이지 번호, 표 순번(No.), 섹션 번호(03, 04 등) 같은 텍스트 노이즈는 무시하고 핵심 의미만 전달하세요.\n2. '오류35'와 같이 인덱스가 붙은 단어를 그대로 내뱉지 마세요.\n3. 지침서 내용을 인용할 때는 '지침에 따르면 ~입니다'라고 풀어서 설명하세요.\n4. 내용이 방대하다면 불필요한 수식어는 빼고 불렛 포인트(•)를 사용하여 요약하세요.";
    let vectorContext = "";

    // [추가] 지침서 기반 답변을 위한 벡터 검색 로직
    if (type === 'point_search') {
        // 1. 질문의 임베딩 생성
        const embedRes = await fetch(`${EMBED_URL}?key=${GEMINI_API_KEY}`, {
            method: 'POST',
            body: JSON.stringify({ 
                content: { parts: [{ text: prompt }] }, 
                task_type: "retrieval_query"
            })
        });
        const embedData = await embedRes.json();
        const queryEmbedding = embedData.embedding.values;

        // 2. Supabase RPC 호출하여 유사 지식 검색
        const searchRes = await fetch(`${SUPABASE_URL}/rest/v1/rpc/match_pdf_knowledge`, {
            method: 'POST',
            headers: {
                'apikey': SUPABASE_KEY,
                'Authorization': `Bearer ${SUPABASE_KEY}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                query_embedding: queryEmbedding,
                match_threshold: 0.5,
                match_count: 3
            })
        });
        const matchedKnowledge = await searchRes.json();

        if (matchedKnowledge && matchedKnowledge.length > 0) {
            vectorContext = "\n\n[학습된 지침서 내용]\n" + matchedKnowledge.map(k => k.content).join("\n---\n");
            systemInstruction += "제공된 '학습된 지침서 내용'을 최우선 근거로 사용하여 답변하세요. ";
        }

        systemInstruction += "사용자가 입력한 검색어가 GIS 관련 용어라면 제공된 도면 레이어 정보를 참고하여 답변하고, 만약 일반적인 질문이라면 범용적인 AI로서 답변하세요.";
    } else if (type === 'pdf_summary') {
      systemInstruction += "제공된 PDF 텍스트 내용을 요약하고 사용자의 질문에 답하세요.";
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000);

    const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify({
        contents: [{
          parts: [{
            text: `${systemInstruction}\n\n[도면 레이어 맥락]\n${context || ""}${vectorContext}\n\n[User Query]\n${prompt}`
          }]
        }],
        generationConfig: {
          temperature: 0.7,
          maxOutputTokens: 512,
        }
      }),
    });

    clearTimeout(timeoutId);

    const data = await response.json();
    
    if (data.error) {
      return new Response(
        JSON.stringify({ success: false, error: data.error.message }),
        { 
          headers: { ...corsHeaders, 'Content-Type': 'application/json' },
          status: response.status // Google API의 실제 상태 코드(429 등)를 그대로 전달
        }
      );
    }

    const aiAnswer = data.candidates[0].content.parts[0].text;
    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: aiAnswer,
        model: "gemini-2.5-flash"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 400 
      }
    );
  }
})
