import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
// [유지] 사용자 요청에 따라 최신 모델인 gemini-2.5-flash-lite 유지
const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // [추가] AI가 사용할 수 있는 도구(함수) 정의
    const tools = [
      {
        function_declarations: [
          {
            name: "query_table_data",
            description: "지침서 및 매뉴얼(pdf_knowledge) 정보를 조회합니다. 검색어는 문장 형태가 아닌 핵심 명사 위주로 띄어쓰기를 포함하여 입력하세요. (예: '네트워크RTK 기준' 등)",
            parameters: {
              type: "OBJECT",
              properties: {
                search_keyword: { type: "STRING", description: "검색 키워드 (예: 네트워크RTK, 도로대장 등)" }
              },
              required: ["search_keyword"]
            }
          }
        ]
      }
    ];

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = `당신은 '아신테크 GIS 관리 도구'의 지식 검색 에이전트 '아신'입니다.
[데이터 접근 규칙]
1. 웹 클라이언트에서 전달한 [분석 대상 데이터]가 있다면 이를 최우선적으로 사용하여 답변하세요.
2. 제공된 데이터 내에서 정답을 찾을 수 없는 경우, 당신의 내부 지식을 활용하여 답변하되 반드시 해당 내용은 'AI 추론 결과'임을 명시하세요.
3. **근거 제시(필수)**: 모든 정보에는 반드시 해당 문서의 이름과 페이지를 포함한 출처를 명시하세요. 형식: [출처: 파일명 p.페이지].
4. **시각 자료 활용**: 검색된 데이터에 'table_svg_urls'나 'image_urls'가 있다면 해당 URL들을 답변 내에 반드시 [ATTACH_SVG:URL] 또는 [ATTACH_IMG:URL] 태그 형식으로 포함하세요.
3. 정체성 규정: 당신의 이름은 '아신'이지만, 답변 시 자기소개는 절대 하지 마세요. 정보 전달에만 집중하세요.`;

    if (type === 'point_search') {
      systemInstruction += "\n사용자가 입력한 검색어가 GIS 관련 용어라면 제공된 도면 레이어 정보를 참고하여 답변하고, 만약 일반적인 질문이라면 범용적인 AI로서 답변하세요.";
    } else if (type === 'pdf_summary') {
      systemInstruction += `
[요약 및 가시화 전문가 모드]
1. 당신의 역할은 사용자가 검색한 DB 결과물을 시각적으로 정리하는 '포맷터(Formatter)'입니다.
2. 제공된 텍스트 원문을 바탕으로 항목별 요약, 강조, 표(Table) 등을 활용하여 가독성 좋게 정리하세요. 
   특히 원문의 기호(●, ■, ※, ① 등)를 적극적으로 유지하여 지침서 특유의 계층 구조를 보존하세요.
3. **출처 명시**: 각 항목마다 \`[출처: 파일명 p.페이지]\`를 붙여 어떤 근거로 작성되었는지 밝히세요.
4. 시각 자료 활용(필수): 검색된 행(Row)에 'table_svg_urls' 또는 'image_urls' 배열 데이터가 존재한다면, 해당 URL들을 답변 내에 반드시 포함시켜야 합니다.
   - 표(SVG)는 [ATTACH_SVG:URL] 형식으로 출력하세요.
   - 그림(WebP)은 [ATTACH_IMG:URL] 형식으로 출력하세요.`;
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000);

    // [핵심] 에이전틱 루프 구현: AI가 도구를 다 쓸 때까지 최대 5회 반복
    let contents = [{
      role: "user",
      parts: [{ text: `${systemInstruction}\n\n[도면 레이어 맥락]\n${context || ""}\n\n[User Query]\n${prompt}` }]
    }];
    let finalAnswer = "";
    let lastRawData = [];
    let loopCount = 0;
    let data = null; // data 변수를 loop 외부에서 접근 가능하도록 상단에 선언

    while (loopCount < 5) {
      const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        signal: controller.signal,
        body: JSON.stringify({
          contents: contents,
          tools: tools,
          generationConfig: { temperature: 0.2, maxOutputTokens: 2048 }
        }),
      });

      data = await response.json(); // const를 제거하여 외부 변수 data를 업데이트
      if (data.error) throw new Error(data.error.message);

      const candidate = data.candidates?.[0];
      const part = candidate?.content?.parts?.[0];

      // 1. 함수 호출 요청이 있는 경우
      if (part?.functionCall) {
        const call = part.functionCall;
        let result = [];

        if (call.name === "query_table_data") {
          const { search_keyword } = call.args;
          let url = `${SUPABASE_URL}/rest/v1/pdf_knowledge?select=*`;
          if (search_keyword) {
            // [개선] 공백을 와일드카드(*)로 변환하여 띄어쓰기나 순서가 조금 달라도 검색되도록 유연화
            // 예: "공공 측량 기준" -> "*공공*측량*기준*"
            const fuzzyKeyword = search_keyword.trim().split(/\s+/).join('*');
            url += `&content=ilike.*${encodeURIComponent(fuzzyKeyword)}*`;
          }
          const dbRes = await fetch(url, { headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}` } });
          result = await dbRes.json();
          if (Array.isArray(result)) lastRawData.push(...result); // 데이터 누락 방지를 위한 누적 저장
        }

        // 대화 기록에 추가하고 루프 계속
        contents.push(candidate.content); // 모델의 function_call 추가
        contents.push({
          role: "function",
          parts: [{ functionResponse: { name: call.name, response: { content: result } } }]
        });
        loopCount++;
      } 
      // 2. 최종 텍스트 답변이 온 경우
      else if (part?.text) {
        finalAnswer = part.text;
        break;
      } 
      else {
        break;
      }
    }

    clearTimeout(timeoutId);

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer || "답변을 생성할 수 없습니다.",
        rawData: JSON.stringify(lastRawData),
        model: data.model_version || "gemini-2.5-flash-lite"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (30초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: isTimeout ? 504 : 400 
      }
    );
  }
})
