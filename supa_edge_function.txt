import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
// [수정] 실제 존재하는 최신 안정화 모델인 gemini-1.5-flash로 변경 (400 에러 해결 핵심)
const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = `당신은 '아신테크 GIS 관리 도구'의 지식 검색 에이전트 '아신'입니다.
[데이터 접근 규칙]
1. 웹 클라이언트에서 전달한 [분석 대상 데이터]가 질문의 의도와 부합하는지 먼저 분석하세요.
2. **데이터 활용**: 검색된 내용이 질문과 연관성이 높다면 이를 요약하여 답변하세요. 만약 데이터가 질문의 의도와 너무 다르거나 정보가 부족하다면, 당신의 전문적인 지식을 활용하여 답변을 보완하거나 대안을 제시하세요.
3. **상태 알림**: 당신의 일반 지식을 활용해 답변을 보완한 경우, 답변 하단에 "지침 DB 내 관련 정보가 부족하여 AI의 일반 지식을 바탕으로 설명해 드립니다."라고 명시하세요.
4. **근거 제시**: DB에서 발췌한 정보에는 반드시 출처를 명시하세요. 형식: [출처: 파일명 p.페이지].
5. **시각 자료**: 데이터에 URL이 포함되어 있다면 [ATTACH_SVG:URL] 또는 [ATTACH_IMG:URL] 태그를 반드시 유지하세요.`;

    if (type === 'point_search') {
      systemInstruction += "\n당신은 도면 레이어와 DB 내용을 분석하여 사용자에게 최적의 답변을 제공합니다. DB에 근거가 없더라도 가능한 전문적인 조언을 제공하되 출처가 없는 정보임을 밝히세요.";
    } else if (type === 'pdf_summary') {
      systemInstruction += `
[요약 및 가시화 전문가 모드]
1. 당신은 고도로 숙련된 '데이터 포맷터(Formatter)'입니다.
2. 제공된 텍스트 원문을 바탕으로 가독성이 뛰어난 레이아웃(표, 리스트, 강조 등)을 설계하여 예쁘게 정리하세요. 
   특히 원문의 기호(●, ■, ※, ① 등)를 적극적으로 유지하여 지침서 특유의 계층 구조를 보존하세요.
3. **출처 명시**: 각 항목마다 \`[출처: 파일명 p.페이지]\`를 붙여 어떤 근거로 작성되었는지 밝히세요.
4. 시각 자료 활용(필수): 검색된 행(Row)에 'table_svg_urls' 또는 'image_urls' 배열 데이터가 존재한다면, 해당 URL들을 답변 내에 반드시 포함시켜야 합니다.
   - 표(SVG)는 [ATTACH_SVG:URL] 형식으로 출력하세요.
   - 그림(WebP)은 [ATTACH_IMG:URL] 형식으로 출력하세요.`;
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000);

    const contents = [{
      role: "user",
      parts: [{ text: `${systemInstruction}\n\n[도면 레이어 맥락]\n${context || ""}\n\n[User Query]\n${prompt}` }]
    }];

    // [수정] 에이전트 루프 제거, 단일 호출 방식으로 변경
    const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify({
        contents: contents,
        generationConfig: { temperature: 0.2, maxOutputTokens: 2048 }
      }),
    });

    const data = await response.json();
    if (data.error) {
      console.error("Gemini API Error:", data.error);
      throw new Error(`Gemini API: ${data.error.message}`);
    }

    const finalAnswer = data.candidates?.[0]?.content?.parts?.[0]?.text || "";

    clearTimeout(timeoutId);

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer || "답변을 생성할 수 없습니다.",
        model: data.model_version || "gemini-2.5-flash-lite"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (30초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: isTimeout ? 504 : 400 
      }
    );
  }
})
