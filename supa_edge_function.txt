import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_ANON_KEY = Deno.env.get("SUPABASE_ANON_KEY");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    let searchContext = context || "";

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = `당신은 지하시설물 및 도로대장 구축 분야의 고도로 숙련된 전문 기술 컨설턴트입니다.
[데이터 접근 규칙]
1. **응답 스타일**: 인사말, 자기소개, 부연설명 없이 사용자의 질문에 대한 핵심 기술 답변만 즉시 출력하세요.
2. **데이터 활용**: 제공된 [분석 대상 데이터]를 바탕으로 답변하되, 데이터에 포함된 URL 정보가 있다면 이를 활용하여 답변 중간에 시각 자료를 배치하세요.
3. **출처 및 근거 생략**: [출처: ...]와 같은 텍스트나 파일명 정보는 답변에 절대 포함하지 마세요.
4. **시각 자료 극대화**: 데이터에 표(SVG)나 이미지(WebP) URL이 포함되어 있다면, 해당 URL을 답변의 맥락에 맞게 [ATTACH_SVG:URL] 또는 [ATTACH_IMG:URL] 태그로 반드시 변환하여 삽입하세요.
5. **수식 표현**: 분수, 제곱근, 거듭제곱 등 복잡한 수학 공식은 반드시 LaTeX 형식($$ ... $$)을 사용하여 작성하세요.
6. **GIS 문법 안내**: 사용자가 검색 문법이나 사용법을 물어보면 아래 규칙을 마크다운 강조(**...**)를 사용하여 설명하세요. (강조 시 파란색으로 표시됩니다)
   - **[레이어명]📋** : 레이어 데이터 리스트 출력
   - **[레이어명]�** : 지도에 위치 표시
   - **[레이어명][거리]📋** : 레이어 총 연장(길이) 계산
   - **[레이어명][사진]📋** : 사진 매칭 누락 분석
   - **지점A ~ 지점B [거리]📋** : 지점 간 경로 거리 계산
   - **[레이어명][거리]>숫자📋** : 간격 이상 구간 분석
   - **키워드1 키워드2📍** : AND 검색 (두 단어 모두 포함)
   - **키워드1 & 키워드2📍** : OR 검색 (둘 중 하나 포함)`;

    if (type === 'point_search') {
      systemInstruction += "\n도면 레이어 정보를 분석하여 실무 가이드를 제공하세요.";
    } else if (type === 'translate_gis') {
      systemInstruction = `당신은 사용자의 자연어 요청을 GIS 전용 검색 문법으로 변환하는 전문 번역 에이전트입니다.

[핵심 미션]
1. **레이어 vs 포인트 구분**: 사용자가 언급한 대상이 **[대상 레이어 목록]에 정확히 일치하는 이름일 경우에만** 대괄호 \`[레이어명]\`을 사용하세요.
2. **복합 검색**: "X 레이어에서 Y를 찾아줘"와 같은 요청은 \`[X] Y📍\` 형식으로 변환하세요.
3. 시설물 명칭이라도 목록에 없는 이름은 절대 대괄호를 씌우지 마세요. (예: "이토변"이 목록에 없으면 \`이토변📍\`)
4. **제외 검색(NOT)**: "X를 제외한", "빼고" 등의 요청은 \`!\` 연산자를 사용하세요. (예: "하단 빼고 제수변" -> \`제수변 !하단📍\`)
5. **교차 분석**: 특정 선 레이어 간의 교차점을 분석합니다.
   - \`[A][교차]📍\` : A와 교차되는 **모든** 선 레이어 검색
   - \`[A][B][교차]📍\` : A와 B **두 레이어 간의** 교차점만 검색
   - \`[A][B][C][교차]📍\` : A와 B, C 레이어들이 만나는 지점 분석 (다중 레이어 지원)
6. **정보 조회**: 특정 포인트의 좌표(TM, 경위도, Z)를 요청하는 경우 '포인트명[좌표]📋' 문법을 사용하세요. (예: "260424-04 좌표 알려줘" -> 260424-04[좌표]📋, "A와 B의 좌표" -> A&B[좌표]📋)
7. **복합 필터링**: 검색어 필터링(AND, OR, NOT) 요청 시 아래 문법 규칙 7~12번을 적극 활용하세요.

[문법 규칙]
1. [레이어명]📋 : 레이어 데이터 리스트 출력 (분석, 정리, 계산, 리스트 등)
2. [레이어명]📍 : 레이어 객체 지도에 마커 표시 (위치, 어디야, 보여줘 등)
3. [레이어명][거리]📋 : 레이어의 총 연장(길이) 계산
4. [레이어명][사진]📋 : 레이어의 사진/메모 매칭 누락 분석
5. 지점A ~ 지점B [거리]📋 : 두 지점 간의 경로 거리 계산
6. [레이어명][거리]>숫자📋 : 포인트 간 간격이 숫자(m)보다 큰 구간 분석
7. [레이어1][레이어2...][교차]📋 : 선 레이어 간 교차점 분석 (여러 레이어 지정 가능)
8. 키워드1 키워드2📍 : **AND 검색** (공백 사용, 두 키워드 모두 포함된 포인트)
9. 키워드1 & 키워드2📍 : **OR 검색** (& 사용, 둘 중 하나라도 포함된 포인트)
10. 키워드1 !키워드2📍 : **NOT 검색** (! 사용, 키워드1은 포함하되 키워드2는 제외)
10. [레이어명] 키워드📍 : 특정 레이어 범위 내에서 키워드 검색 (가장 권장되는 정밀 검색 방식)
11. 포인트명[좌표]📋 : 포인트의 상세 좌표(TM X, TM Y, 경위도, Z) 및 속성 조회

[주의] 답변은 정의된 문법 문자열 하나만 출력하세요. 자바스크립트 코드를 직접 작성하지 마세요. 부연 설명은 절대 금지이며, 반드시 마지막은 📍 또는 📋로 끝나야 합니다.

[대상 레이어 목록] (현재 도면에 존재하는 레이어들)
${searchContext}`;
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 40000);

    const contents = [{
      role: "user",
      parts: [{ text: `${systemInstruction}\n\n[분석 대상 데이터]\n${searchContext}\n\n[User Query]\n${prompt}` }]
    }];

    // [수정] 에이전트 루프 제거, 단일 호출 방식으로 변경
    const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify({
        contents: contents,
        generationConfig: { temperature: 0.2, maxOutputTokens: 2048 }
      }),
    });

    const data = await response.json();
    if (data.error) {
      console.error("Gemini API Error:", data.error);
      throw new Error(`Gemini API: ${data.error.message}`);
    }

    const finalAnswer = data.candidates?.[0]?.content?.parts?.[0]?.text || "";

    clearTimeout(timeoutId);

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer || "답변을 생성할 수 없습니다.",
        model: data.model_version || "gemini-2.5-flash-lite"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (30초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: isTimeout ? 504 : 400 
      }
    );
  }
})
