import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const API_BASE = "https://generativelanguage.googleapis.com/v1beta";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const body = await req.json().catch(() => ({}));
    const { prompt, context, type } = body;

    if (type === 'list_models') {
      if (!GEMINI_API_KEY) throw new Error("GEMINI_API_KEY 환경 변수가 설정되지 않았습니다.");
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: listResponse.status }
        );
      }
      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    if (!prompt) {
      return new Response(
        JSON.stringify({ success: false, error: "요청바디에 'prompt' 값이 누락되었습니다." }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
      );
    }

    let searchContext = context || "";
    let systemInstructionText = `당신은 지하시설물 및 도로대장 구축 분야의 고도로 숙련된 전문 기술 컨설턴트입니다.
[데이터 접근 규칙]
1. **GIS 문법 안내**: 사용자가 검색 문법이나 사용법을 물어보면 아래 규칙을 마크다운 강조(**...**)를 사용하여 설명하세요.
   - **[레이어명]📋** : 레이어 데이터 리스트 출력
   - **[레이어명]📍** : 지도에 위치 표시
   - **[레이어명][거리]📋** : 레이어 총 연장(길이) 계산
   - **[레이어명][사진]📋** : 사진 매칭 누락 분석
   - **지점A ~ 지점B [거리]📋** : 지점 간 도상/경로 거리 계산
   - **키워드1 키워드2📍** : AND 검색 (두 단어 모두 포함)
   - **키워드1 & 키워드2📍** : OR 검색 (둘 중 하나 포함)`;

    if (type === 'point_search') {
      systemInstructionText += "\n도면 레이어 정보를 분석하여 실무 가이드를 제공하세요.";
    } else if (type === 'translate_gis') {
      systemInstructionText = `당신은 사용자의 자연어 요청을 GIS 전용 검색 문법으로 정확히 변환하는 전문 번역 에이전트입니다. 다른 설명은 절대 배제하고 오직 변환된 문법 결과만 반환하세요.

[핵심 미션]
1. 검색어 정규화: 수치 뒤의 단위('mm', 'cm' 등)는 제거하고 숫자만 남기세요. 명칭과 숫자 사이의 공백도 제거하여 결합하세요. (예: "제수변 150mm" -> 제수변150)
2. 레이어 vs 포인트 구분: 사용자가 언급한 대상이 [대상 레이어 목록]에 정확히 일치하는 이름일 경우에만 대괄호 [레이어명]을 사용하세요. 목록에 없는 이름은 절대 대괄호를 씌우지 마세요.
3. 복합 검색: "X 레이어에서 Y를 찾아줘"와 같은 요청은 [X] Y📍 형식으로 변환하세요.

[기본 문법 규칙]
1. [레이어명]📋 : 리스트 및 분석 출력
2. [레이어명]📍 : 지도 위치 표시
3. [레이어명][거리]📋 : 총 연장(길이) 계산
4. [레이어명][사진]📋 : 사진/메모 매칭 분석
5. A ~ B [거리]📋 : 지점 간 도상 거리
6. [레이어명][거리]>X📋 : 간격 오차 분석
7. [레이어1][레이어2...][교차]📋 : 교차점 분석
8. 키워드1 키워드2📍 : **AND 검색**
9. 키워드1 & 키워드2📍 : **OR 검색**
10. 키워드1 !키워드2📍 : **NOT 검색**

[★ 다중 작업 연산자 기호(&& vs +) 판단 규칙 ★]
문장에 여러 레이어나 복수 요청이 결합되어 있을 때, 사용자의 논리적 의도를 파악하여 기호를 선택하세요.

1. '+' 연산자 (산술적 합산):
   - 조건: 두 개 이상의 데이터를 하나의 값으로 '더하기', '합산'할 때 사용. (예: "A와 B의 연장 합계")
   - 올바른 변환: [A][거리] + [B][거리]📋

2. '&&' 연산자 (독립적인 다중 결과 병렬 출력):
   - 조건: 서로 다른 항목을 '각각', '동시에' 보여달라고 요청할 때 사용.
   - 올바른 변환: [A][거리]📋 && [B]📋

[대상 레이어 목록] (현재 도면에 존재하는 레이어들)
${searchContext}`;
    } else if (type === 'narrate_gis') {
      systemInstructionText = `당신은 분석된 GIS 결과 데이터를 바탕으로 사용자에게 결과를 친절하게 서술하는 AI 에이전트입니다.

[절대 규칙]
1. [도면 데이터 검색 결과 요약] 텍스트에 있는 실제 수치(거리, 수량 등)와 팩트만을 사용하여 답변하세요.
2. 텍스트에 없는 수치나 정보를 절대 지어내지 마세요 (할루시네이션 절대 금지).
3. 답변은 반드시 아래의 JSON 객체 형식으로만 출력해야 하며, 앞뒤에 어떤 인사말이나 부연 설명도 추가하지 마세요.

[출력 JSON 형식]
{
  "intent": "사용자 질문 의도 요약 (한 문장)",
  "applied_grammar": "적용된 문법 기호 요약",
  "narrative_answer": "계산 결과를 친절하게 설명하는 서술형 답변 (반드시 실제 요약 텍스트의 수치를 인용할 것)"
}`;
    }

    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 40000);

    const apiPayload = {
      contents: [{
        role: "user",
        parts: [{ text: `[분석 대상 데이터]\n${searchContext}\n\n[User Query]\n${prompt}` }]
      }],
      systemInstruction: {
        parts: [{ text: systemInstructionText }]
      },
      generationConfig: { 
        temperature: 0.05, 
        maxOutputTokens: 1024 
      }
    };

    const API_URL = `${API_BASE}/models/gemini-2.5-flash:generateContent?key=${GEMINI_API_KEY}`;

    const response = await fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify(apiPayload),
    });
    
    const data = await response.json();
    clearTimeout(timeoutId);

    if (data.error) {
      console.error("Gemini API Error Detail:", data.error);
      return new Response(
        JSON.stringify({ success: false, error: `Gemini API 에러: ${data.error.message}` }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 502 }
      );
    }

    let finalAnswer = data.candidates?.[0]?.content?.parts?.[0]?.text || "";
    
    // JSON 모드일 경우 순수 텍스트만 리턴
    if (type === 'narrate_gis') {
      finalAnswer = finalAnswer.replace(/^```json\s*/i, "").replace(/```$/, "").trim();
    }

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer || "답변을 생성할 수 없습니다.",
        model: "gemini-2.5-flash"
      }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (40초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' }, 
        status: isTimeout ? 504 : 500
      }
    );
  }
})
