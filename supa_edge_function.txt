import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

// 환경 변수(Secrets)에서 안전하게 가져오도록 변경
const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL");
const SUPABASE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); // 서비스 롤 키 사용 권장

const API_BASE = "https://generativelanguage.googleapis.com/v1beta";
const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent`;
// 사용자님의 모델 리스트 35번 항목(gemini-embedding-001)으로 모델 변경
const EMBED_URL = `${API_BASE}/models/gemini-embedding-001:embedContent`;

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  // CORS 프리플라이트 요청 처리
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { prompt, context, type } = await req.json();

    // [수정] 실시간 임베딩 생성 기능 복구
    if (type === 'get_embedding') {
      const embedRes = await fetch(`${EMBED_URL}?key=${GEMINI_API_KEY}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          content: { parts: [{ text: prompt }] }, 
          task_type: "RETRIEVAL_DOCUMENT"
        })
      });
      const embedData = await embedRes.json();
      
      if (embedData.error) {
        throw new Error(`[Embedding API] ${embedData.error.message}`);
      }

      return new Response(
        JSON.stringify({ success: true, embedding: embedData.embedding?.values || null }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // [추가] 사용 가능한 모델 리스트 조회 기능
    if (type === 'list_models') {
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { 
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: listResponse.status 
          }
        );
      }

      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // 시스템 프롬프트 설정 (학습형 AI의 정체성 부여)
    let systemInstruction = "당신은 '아신테크 GIS 관리 도구'의 전문 어시스턴트입니다. [1000자 이내 완결]과 [가독성 최적화]를 절대 원칙으로 합니다. \n\n[답변 원칙]\n1. 반드시 (~니다, ~요) 체로 답변하고 1000자 이내에서 문장을 완벽히 종결하세요.\n2. 지침서 원문의 텍스트가 표 형식임에도 엉켜있다면, 지능적으로 판단하여 마크다운 표(|---|)로 재구성하여 답변하세요.\n3. 한국어 맞춤법을 엄격히 준수하며 마침표(.)와 쉼표(,) 뒤에는 반드시 한 칸을 띄우세요.\n4. 핵심 용어는 **굵게** 표시하고 중요한 문장은 <u>밑줄</u>을 사용하세요.\n5. 페이지 번호 등 텍스트 노이즈는 제거하고 가급적 불렛 포인트(•)를 활용하세요.";
    let vectorContext = "";

    // [추가] 지침서 기반 답변을 위한 벡터 검색 로직
    if (type === 'point_search') {
        // 1. [복구] 질문의 임베딩 생성
        const embedRes = await fetch(`${EMBED_URL}?key=${GEMINI_API_KEY}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ 
                content: { parts: [{ text: prompt }] }, 
                task_type: "RETRIEVAL_QUERY"
            })
        });
        const embedData = await embedRes.json();

        if (embedData.error) {
            throw new Error(`[Embedding API] ${embedData.error.message}`);
        }
        const queryEmbedding = embedData.embedding?.values;

        // 2. Supabase RPC 호출하여 유사 지식 검색 (지침서 DB 탐색)
        const searchRes = await fetch(`${SUPABASE_URL}/rest/v1/rpc/match_pdf_knowledge`, {
            method: 'POST',
            headers: {
                'apikey': SUPABASE_KEY,
                'Authorization': `Bearer ${SUPABASE_KEY}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                query_embedding: queryEmbedding,
                match_threshold: 0.5,
                match_count: 5 // 더 풍부한 맥락을 위해 검색 결과 수를 늘림
            })
        });

        let matchedKnowledge = [];
        if (searchRes.ok) {
            matchedKnowledge = await searchRes.json();
        } else {
            const errorText = await searchRes.text();
            console.error("Supabase RPC Error:", errorText);
            // DB 검색 실패 시 AI가 일반 지식으로라도 답변할 수 있게 vectorContext를 비우고 계속 진행
        }

        if (matchedKnowledge && matchedKnowledge.length > 0) {
            vectorContext = "\n\n[학습된 지침서 내용]\n" + matchedKnowledge.map(k => k.content).join("\n---\n");
            systemInstruction += "제공된 '학습된 지침서 내용'을 최우선 근거로 사용하여 답변하세요. ";
        }

        systemInstruction += "사용자가 입력한 검색어가 GIS 관련 용어라면 제공된 도면 레이어 정보를 참고하여 답변하고, 만약 일반적인 질문이라면 범용적인 AI로서 답변하세요.";
    } else if (type === 'pdf_summary') {
      systemInstruction += "제공된 PDF 텍스트 내용을 요약하고 사용자의 질문에 답하세요.";
    }

    // [추가] 30초 타임아웃 설정 (Gate가 무한정 열려있는 것 방지)
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000);

    const response = await fetch(`${API_URL}?key=${GEMINI_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify({
        contents: [{
          parts: [{
            text: `${systemInstruction}\n\n[도면 레이어 맥락]\n${context || ""}${vectorContext}\n\n[User Query]\n${prompt}`
          }]
        }],
        generationConfig: {
          temperature: 0.7,
          maxOutputTokens: 2048, // 1000자 이상의 충분한 토큰을 확보하여 한국어 답변 끊김 방지
        }
      }),
    });

    clearTimeout(timeoutId);

    const data = await response.json();
    
    if (data.error) {
      return new Response(
        JSON.stringify({ success: false, error: data.error.message }),
        { 
          headers: { ...corsHeaders, 'Content-Type': 'application/json' },
          status: response.status // Google API의 실제 상태 코드(429 등)를 그대로 전달
        }
      );
    }

    // 응답 데이터 구조 안전하게 접근
    const aiAnswer = data.candidates?.[0]?.content?.parts?.[0]?.text || "답변을 생성할 수 없습니다.";
    
    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: aiAnswer,
        model: data.model_version || "gemini-2.5-flash"
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
      }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (30초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: isTimeout ? 504 : 400 
      }
    );
  }
})
