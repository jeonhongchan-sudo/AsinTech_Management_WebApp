import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

const GEMINI_API_KEY = Deno.env.get("GEMINI_API_KEY");
const API_BASE = "https://generativelanguage.googleapis.com/v1beta";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const body = await req.json().catch(() => ({}));
    const { prompt, context, type } = body;

    // [수정] 1. 모델 리스트 조회 시, 하위 복잡한 로직을 거치지 않고 즉시 반환하도록 완전 분리
    if (type === 'list_models') {
      if (!GEMINI_API_KEY) throw new Error("GEMINI_API_KEY 환경 변수가 설정되지 않았습니다.");
      const listResponse = await fetch(`${API_BASE}/models?key=${GEMINI_API_KEY}`);
      const listData = await listResponse.json();
      
      if (listData.error) {
        return new Response(
          JSON.stringify({ success: false, error: listData.error.message }),
          { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: listResponse.status }
        );
      }
      return new Response(
        JSON.stringify({ success: true, models: listData.models }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      );
    }

    // 2. 일반 텍스트 분석/번역 요청인데 prompt가 없는 경우 가차없이 클라이언트 에러 반환
    if (!prompt) {
      return new Response(
        JSON.stringify({ success: false, error: "요청바디에 'prompt' 값이 누락되었습니다." }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
      );
    }

    let searchContext = context || "";

    // 시스템 프롬프트 정의
    let systemInstructionText = `당신은 지하시설물 및 도로대장 구축 분야의 고도로 숙련된 전문 기술 컨설턴트입니다.
[데이터 접근 규칙]
1. **GIS 문법 안내**: 사용자가 검색 문법이나 사용법을 물어보면 아래 규칙을 마크다운 강조(**...**)를 사용하여 설명하세요. (강조 시 파란색으로 표시됩니다)
   - **[레이어명]📋** : 레이어 데이터 리스트 출력
   - **[레이어명]📍** : 지도에 위치 표시
   - **[레이어명][거리]📋** : 레이어 총 연장(길이) 계산
   - **[레이어명][사진]📋** : 사진 매칭 누락 분석
   - **지점A ~ 지점B [거리]📋** : 지점 간 도상/경로 거리 계산
   - **키워드1 키워드2📍** : AND 검색 (두 단어 모두 포함)
   - **키워드1 & 키워드2📍** : OR 검색 (둘 중 하나 포함)`;

    if (type === 'point_search') {
      systemInstructionText += "\n도면 레이어 정보를 분석하여 실무 가이드를 제공하세요.";
    } else if (type === 'translate_gis') {
      systemInstructionText = `당신은 사용자의 자연어 요청을 GIS 전용 검색 문법으로 변환하는 전문 번역 에이전트입니다.
[핵심 미션]
1. **검색어 정규화**: 수치 뒤의 단위('mm', 'cm' 등)는 제거하고 숫자만 남기세요. 명칭과 숫자 사이의 공백도 제거하여 결합하세요. (예: "제수변 150mm" -> \`제수변150\`)
2. **레이어 vs 포인트 구분**: 사용자가 언급한 대상이 [대상 레이어 목록]에 정확히 일치하는 이름일 경우에만 대괄호 \`[레이어명]\`을 사용하세요.
3. **복합 검색**: "X 레이어에서 Y를 찾아줘"와 같은 요청은 \`[X] Y📍\` 형식으로 변환하세요.
4. 시설물 명칭이라도 목록에 없는 이름은 절대 대괄호를 씌우지 마세요.
5. **제외 검색(NOT)**: "하단 빼고 제수변" -> \`제수변 !하단📍\`
6. **교차 분석**: 선 레이어 간 교차점을 분석합니다. \`[A][교차]📍\`, \`[A][B][교차]📍\`
6. **정보 조회**: 포인트 상세 좌표 요청 시 '포인트명[좌표]📋' 문법 사용
7. **복합 기능 결합**: \`제수변 !하단[사진]📋\` 형태로 자유롭게 결합
8. **다중 작업 수행(중요)**: 합산(+) 혹은 다중 질문(&&) 구분자 사용

[문법 규칙]
1. [레이어명]📋 : 레이어 데이터 리스트 출력
2. [레이어명]📍 : 레이어 객체 지도에 마커 표시
3. [레이어명][거리]📋 : 레이어의 총 연장 계산
4. [레이어명][사진]📋 : 레이어의 사진 매칭 누락 분석
5. 지점A ~ 지점B [거리]📋 : 두 지점 간의 도상 거리 계산
6. [레이어명][거리]>숫자📋 : 포인트 간 간격 분석
7. [레이어1][레이어2...][교차]📋 : 선 레이어 간 교차점 분석
8. 키워드1 키워드2📍 : AND 검색
9. 키워드1 & 키워드2📍 : OR 검색
10. 키워드1 !키워드2📍 : NOT 검색
11. [레이어명] 키워드📍 : 특정 레이어 내 상세 검색
12. 포인트명[좌표]📋 : 포인트의 상세 좌표 및 속성 조회

[주의] 답변은 정의된 문법 문자열 하나만 출력하세요. 자바스크립트 코드를 직접 작성하지 마세요. 부연 설명은 절대 금지이며, 반드시 마지막은 📍 또는 📋로 끝나야 합니다.

[대상 레이어 목록] (현재 도면에 존재하는 레이어들)
${searchContext}`;
    }

    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 40000);

    // [수정] 3. Gemini API 공식 규격에 맞춰 systemInstruction을 별도 파라미터로 추출하여 Body 구성
    const apiPayload = {
      contents: [{
        role: "user",
        parts: [{ text: `[분석 대상 데이터]\n${searchContext}\n\n[User Query]\n${prompt}` }]
      }],
      systemInstruction: {
        parts: [{ text: systemInstructionText }]
      },
      generationConfig: { 
        temperature: 0.1, // 번역 정확도를 위해 값 조절
        maxOutputTokens: 1024 
      }
    };

    const API_URL = `${API_BASE}/models/gemini-2.5-flash-lite:generateContent?key=${GEMINI_API_KEY}`;

    const response = await fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      signal: controller.signal,
      body: JSON.stringify(apiPayload),
    });

    const data = await response.json();
    clearTimeout(timeoutId);

    if (data.error) {
      console.error("Gemini API Error Detail:", data.error);
      return new Response(
        JSON.stringify({ success: false, error: `Gemini API 에러: ${data.error.message}` }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 502 } // 502 Bad Gateway로 명확히 분리
      );
    }

    const finalAnswer = data.candidates?.[0]?.content?.parts?.[0]?.text || "";

    return new Response(
      JSON.stringify({ 
        success: true, 
        answer: finalAnswer.trim() || "답변을 생성할 수 없습니다.",
        model: "gemini-2.5-flash-lite"
      }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
    );

  } catch (error) {
    console.error("Edge Function Runtime Error:", error);
    const isTimeout = error.name === 'AbortError';
    
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: isTimeout ? "AI 응답 시간이 초과되었습니다. (40초)" : error.message 
      }),
      { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' }, 
        status: isTimeout ? 504 : 500 // 단순 문법/런타임 에러는 500 내부 서버 에러로 표기
      }
    );
  }
})
